build library "libscripal.a" from all objects (Poco + scripal)

ar -rc libScripal.a *.o

Now write a test program:

**********************************************

#include <stdio.h>
#include <scripal.hpp>

int main(int argc, char **argv) {
  Scripal::init();
      
  std::cout << "version: " << Scripal::VERSION << "\n\n";
  
  // set time measurement on
  Scripal::config.measureTime = true;
  
  // scripal object
  auto obj = new Scripal::TFScripal("match ('test'); end");;
  obj->match("test");      

	std::cout << "results: " << obj->results.size() << " , '" << obj->results.results[0] << "' , pos: " << obj->results.positions[0] << "\n\n";
	return 0;
}
